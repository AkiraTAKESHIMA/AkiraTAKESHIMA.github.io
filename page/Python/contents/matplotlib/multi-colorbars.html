複数のカラーバー
    Top > Python > 作図 (Matplotlib)
    基本的な実装例 
    import numpy as np
import matplotlib.pyplot as plt
import matplotlib as mpl

# Make data for heatmap
nx, ny = 100, 100
x = np.cos(np.linspace(0, np.pi*8, nx))
y = np.sin(np.linspace(0, np.pi*8, ny))
xx, yy = np.meshgrid(x, y)
data = xx * yy

# Make data for line
lines = []
nLine = 5
for iLine in range(5):
    x = np.linspace(0, nx-1, 1000)
    y = np.sin(x) + ny*(iLine+0.5)/nLine
    v = iLine
    lines.append((x, y, v))

# Prepare ScalarMappable and colormap for line
sm_line = mpl.cm.ScalarMappable(norm=norm_line, cmap=cmap_line)
sm_line.set_array([])
cmap_line = plt.cm.coolwarm
norm_line = mpl.colors.Normalize(vmin=0, vmax=nLine-1)

# Plot
fig, ax = plt.subplots()

im_heat = ax.imshow(data, cmap=plt.cm.viridis, interpolation='nearest')

for (x, y, v) in lines:
    ax.plot(x, y, color=cmap_line(norm_line(v)))

# Colorbar of line
cbar_line = fig.colorbar(
    sm_line, ax=ax, aspect=40, pad=0.08, label='Line'
)

# Colorbar of heatmap
cbar_heat = fig.colorbar(
    im_heat, ax=ax, aspect=40, pad=0.04, label='Heatmap'
)

plt.show()
    　ポイントは 　・Line plotのカラーバーの描画にはScalarMappableを使う 　注意点は以下の2点． 　1. カラーバーは後から描いたものが図の近くに配置される 　　順番を入れ替えるには描画順を変えるか，カラーバー用の軸を別途定義する（後述）． 　　padの値を負の値にするなどして無理矢理変えることも可能ではあるが，調整が面倒でありスマートな方法ではない． 　2. padの基準は，最も図に近いカラーバーの場合は図本体，それ以外は自分よりひとつ内側（図本体に近い側）のカラーバーである 　　この例で言うと，cbar_heatは図本体から0.04, cbar_lineはcbar_heatから0.08の距離に配置される．もっと細かい話をすると，cbar_lineが基準としているのはcbar_heatの軸ではなく，目盛ラベル (0.00とか-0.75とか) の右側となっているようである（推察）．だから例えばcbar_lineでpad=0とすると下図のようになる．
    カラーバーの軸を明示的に定義する
    　データの作成とline plot用のカラーマップの作成までは共通． 　fig.add_axes([left, bottom, width, height])でカラーバーの軸をメインの図と別に定義する． 　メインの図とカラーバーの幅や高さ，2つのカラーバーの太さなどを合わせるために，先にfigsize, left, cbar_hwdithなど諸々の定数を設定している．もちろん各手続きで直接数字を入れてしまっても良い． 　なおimshowだとメインの図の軸の長さが自動的に調整されてしまって分かりづらいので，ここではpcolormeshでヒートマップを描く． 
    figsize = (9, 6)
left, width = 0.1, 0.7
bottom, height = 0.2, 0.7
cbar_hwidth, cbar_vwidth = 0.02, 0.02*figsize[0]/figsize[1]
cbar_hpad, cbar_vpad = 0.05, 0.08

fig, ax = plt.subplots(
    figsize=figsize,
    gridspec_kw=dict(left=left, right=left+width, bottom=bottom, top=bottom+height)
)

# add_axes((left, bottom, width, height))
cax_heat = fig.add_axes((left+width+cbar_hpad, bottom, cbar_hwidth, height))
cax_line = fig.add_axes((left, bottom-cbar_vpad-cbar_vwidth, width, cbar_vwidth))

im_heat = ax.pcolormesh(
    np.linspace(0, nx-1, nx),
    np.linspace(0, ny-1, ny),
    data,
    cmap=plt.cm.viridis,
)

for (x, y, v) in lines:
    ax.plot(x, y, color=cmap_line(norm_line(v)))

cbar_heat = fig.colorbar(im_heat, cax=cax_heat, label='Heatmap')
cbar_line = fig.colorbar(sm_line, cax=cax_line, orientation='horizontal', label='Line')

plt.show()
