    純粋手続き (純粋関数、純粋サブルーチン)
    　次のような手続きを純粋手続きという． 　　・標準入出力やファイルの入出力を行わない 　　・関数の場合は引数と返値，サブルーチンの場合は引数にのみ作用し，他の変数 (モジュール変数など) に影響を与えない 　純粋手続き内で呼び出す手続きは純粋手続きでなければならない． 　また純粋手続きは純粋手続きであることが明示的に示されていなければならない． 　純粋手続きは pure というキーワードで定義する． 　例: 
    program main
  implicit none
  integer :: a, b

  a = 2
  b = 6

  print*, purefunc1(a, b)

contains
!===============================================================
!
!===============================================================
integer pure function purefunc1(a, b) result(c)
  implicit none
  integer, intent(in) :: a, b

  c = a * b

  c = purefunc2(c)

  call puresub1(a, b, c)
end function purefunc1
!===============================================================
!
!===============================================================
integer pure function purefunc2(c) result(x)
  implicit none
  integer, intent(in) :: c

  x = c + 1
end function purefunc2
!===============================================================
!
!===============================================================
pure subroutine puresub1(a, b, c)
  implicit none
  integer, intent(in) :: a, b
  integer, intent(inout) :: c

  if( a < 0 .or. b < 0 ) c = 0
end subroutine puresub1
!===============================================================
!
!===============================================================
end program main
