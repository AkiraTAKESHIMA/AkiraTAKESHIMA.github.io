    Quick Start
    　こういう感じでCの自作関数 doYouDo を呼び出したい． 
    import hello

hello.doYouDo()
>>Hello, I am (null).

hello.doYouDo("an researcher")
>>Hello, I am an researcher.
    Cの関数 what.c 
    #include <stdio.h>

void doYouDo(const char* job){
  printf("Hello, I am %s.\n", job);
}
    Cの関数のラッパー whatWrap.c 
    #include <Python.h>

extern void doYouDo(const char*);

PyObject* what_doYouDo(PyObject* self, PyObject* args, PyObject* kw){
  const char* job = NULL;
  static char* argnames[] = {"job", NULL};
  if(!PyArg_ParseTupleAndKeywords(args, kw, "|s", argnames, &job))
    return NULL;
  doYouDo(job);
  return Py_BuildValue("");
}

static PyMethodDef whatmethods[] = {
  {"doYouDo", (PyCFunction)what_doYouDo, METH_VARARGS | METH_KEYWORDS, ""},
  {NULL, NULL, 0, NULL},
};

static struct PyModuleDef hello = {
  PyModuleDef_HEAD_INIT,
  "hello",
  "Python3 C API Module (Sample 1)",
  -1,
  whatmethods
};

PyMODINIT_FUNC PyInit_hello(void){
  return PyModule_Create(&hello);
};
    コンパイル 
    $ gcc -fPIC -Wall -c -o what.o what.c
$ gcc -fPIC -Wall -c -o whatWrap.o whatWrap.c -I/usr/include/python3.9
$ gcc -fPIC -Wall -shared -o hello.so what.o whatWrap.o
    　これでPythonから呼び出せるCのモジュールhelloができた．最初のサンプルコードのようにして関数を呼び出す．
    解説
    ラッパー 　Pythonから呼び出した際の引数の処理及び戻り値の定義を行う関数を定義する． 
    PyObject* what_doYouDo(PyObject* self, PyObject* args, PyObject* kw){
  const char* job = NULL;
  static char* argnames[] = {"job", NULL};
  if(!PyArg_ParseTupleAndKeywords(args, kw, "|s", argnames, &job))
    return NULL;
  doYouDo(job);
  return Py_BuildValue("");
}
    what_doYouDoの引数 　self, args, kwはそれぞれ関数自身，引数，キーワード．いずれもPyObjec型の構造体で参照渡しらしい． PyArg_ParseTupleAndKeywords 　この手続きで引数とキーワードを解析する．詳細: 　　+ 引数の解釈と値の構築 — Python 3.13.3 ドキュメント - - c.PyArg_ParseTupleAndKeywords 　3つ目の引数 (ここでは|s) はPythonで与えられた引数をCのどの型に変換するかを指定する．代表的なものとしてsは文字列を指すポインタ，iはint型，dはdouble float型など．|以降はキーワード引数，つまり無くても関数が動作する引数．ここでは位置引数 (必須の引数) が無くjobというキーワード引数のみで，これが文字列と想定しているので|sとなる．型についての詳細: 　　+ 引数の解釈と値の構築 — Python 3.13.3 ドキュメント 　4つ目の引数 (argnames) は「The keywords argument is a NULL-terminated array of keyword parameter names...」(ドキュメントより) とのことで，この例のように，キーワードの一覧の後ろにNULLを加えたものを与える． 　Python側で引数の与え方を誤ると，引数の解析に失敗し，PyArg_ParseTupleAndKeywordsがFalseを返す．その場合この例ではNULLを返す (return NULL)． 　また引数jobはキーワード引数ということで，Pythonから値が与えられない場合未定義の状態になってしまうので， 
      const char* job = NULL;
    で定義しておく． 　引数の解析に成功すると，いよいよ関数の本体doYouDoが呼び出される． 
      doYouDo(job);
    　最後に戻り値をPy_BuildValueでPythonオブジェクトに変換して返す． 
      return Py_BuildValue("");
    　この関数のように戻り値を持たない場合は空の文字列を入れる． 　戻り値がある場合はPy_BuildValue(format, value1, value2, ...)のような形で，Cの変数value1, value2, ...をformatに従ってPythonオブジェクトに変換したものを返す．formatは先のPyArg_ParseTupleAndKeywordsと同じ(多分)．詳細: 　　+ 引数の解釈と値の構築 — Python 3.13.3 ドキュメント - c.Py_BuildValue モジュールに入れる手続きのリストアップ 
    static PyMethodDef whatmethods[] = {
  {"doYouDo", (PyCFunction)what_doYouDo, METH_VARARGS | METH_KEYWORDS, ""},
  {NULL, NULL, 0, NULL},
};
    　括弧 {} 内の引数は 　　{手続き名，ラッパーのオブジェクト，引数の与え方，手続きの説明} 　1つ目の「手続き名」はCの関数名と同じである必要は無く，任意に決められるらしい．Python側から呼び出すときにどういう名前でその手続きを呼び出したいか，ということ． 　2つ目は先に定義したラッパー．(PytCFunction)でキャストする． 　3つ目の「引数の与え方」は，この例のようにキーワード引数がある場合はこのように書くが，位置引数のみの場合はMETH_VARARGSだけで良い．引数が無い場合はNULLで良い？　不明，要確認． 　詳細: 　　+ 共通のオブジェクト構造体 (common object structure) — Python 3.13.3 ドキュメント 　最後の{NULL, NULL, 0, NULL}というのは何だか分からないが，終端であることを教える的な役割なのだと思う．知らんけど．単に{NULL}などでも良いらしい． モジュールの定義 　値の細かい説明はあまり重要ではないと思うが，一応書いておく． 
    static struct PyModuleDef hello = {
  PyModuleDef_HEAD_INIT, // 初期化
  "hoge", // モジュールの名前
  "Python3 C API Module (Sample 1)", // 説明
  -1,  // モジュールの状態
  whatmethods  // 先に定義した、モジュールに入れる手続き群
};
    　PyModuleDef型のオブジェクトを作成する． 　必ずPyModuleDef_HEAD_INITで初期化する．名前はモジュール名とは独立に，任意に決められる．もちろん説明も．Python側からはそれぞれ以下のように取得できる． 
    hello.__name__
>>'hoge'

hello.__doc__
>>'Python3 C API Module (Sample 1)'
    　モジュールの状態についてはよく分からないがとりあえず-1で良さそう． モジュールの初期化 　最後に，Pythonでモジュールをimportした際に呼び出される初期化関数を定義する． 　PyModule_Createに，今作ったモジュールを与える． 
    PyMODINIT_FUNC PyInit_hello(void){
  return PyModule_Create(&hello);
};
    コンパイル 
    $ gcc -fPIC -Wall -c -o what.o what.c
$ gcc -fPIC -Wall -c -o whatWrap.o whatWrap.c -I/usr/include/python3.9
$ gcc -fPIC -Wall -shared -o hello.so what.o whatWrap.o
    　1つ目は普通のCのコンパイル． 　2つ目は少し注意．ヘッダファイルPython.hをincludeしているので，そのファイルがあるディレクトリが検索対象に含まれていなければならない．~/.bash_profileなどで予めパスに含まれていない場合は，この例のように-Iオプションでそのディレクトリを追加する． 　3つ目のコマンドで共有ライブラリを作成している． 　以上！　思った以上に手順が多くて大変．
